cmake_minimum_required(VERSION 3.16)
project(SpaceTrainerAlgorithms LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(monitor_algorithms src/Algorithms.cpp)
target_include_directories(monitor_algorithms PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
add_executable(demo src/main.cpp)
target_link_libraries(demo PRIVATE monitor_algorithms)
add_executable(algorithms_tests tests/test_algorithms.cpp)
target_link_libraries(algorithms_tests PRIVATE monitor_algorithms)
enable_testing()
add_test(NAME algorithms_tests COMMAND algorithms_tests)
