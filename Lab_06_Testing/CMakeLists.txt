cmake_minimum_required(VERSION 3.21)
project(SpaceTrainerTesting LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(algorithms_extended_tests tests/algorithms_extended_tests.cpp reference/algorithms/Algorithms.cpp)
target_include_directories(algorithms_extended_tests PRIVATE reference/algorithms)
add_test(NAME algorithms_extended_tests COMMAND algorithms_extended_tests)
option(BUILD_QT_TESTS "Build additional TCP framing tests (requires Qt 6 Test and Network)" OFF)
if(BUILD_QT_TESTS)
    find_package(Qt6 6.2 REQUIRED COMPONENTS Core Network Test)
    set(CMAKE_AUTOMOC ON)
    add_executable(telemetry_qt_tests tests/telemetry_qt_tests.cpp reference/qt/TelemetryServer.cpp reference/qt/TelemetryServer.h)
    target_include_directories(telemetry_qt_tests PRIVATE reference/qt)
    target_link_libraries(telemetry_qt_tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
    add_test(NAME telemetry_qt_tests COMMAND telemetry_qt_tests)
endif()
